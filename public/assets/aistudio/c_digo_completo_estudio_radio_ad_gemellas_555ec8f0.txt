    function playNote(ctx, freq, startTime, duration, type = 'triangle', volume = 0.3, slideTo = null) {
      const osc = ctx.createOscillator();
      const gain = ctx.createGain();
      osc.type = type;
      osc.frequency.setValueAtTime(freq, startTime);
      if (slideTo) {
        osc.frequency.exponentialRampToValueAtTime(slideTo, startTime + duration);
      }

      const master = getMasterGain();
      gain.gain.setValueAtTime(0.001, startTime);
      gain.gain.linearRampToValueAtTime(volume * master, startTime + 0.02);
      gain.gain.exponentialRampToValueAtTime(0.001, startTime + duration);

      osc.connect(gain);
      gain.connect(ctx.destination);
      osc.start(startTime);
      osc.stop(startTime + duration + 0.03);
    }

    function createNoiseBuffer(ctx, durationSec) {
      const bufferSize = ctx.sampleRate * durationSec;
      const buffer = ctx.createBuffer(1, bufferSize, ctx.sampleRate);
      const data = buffer.getChannelData(0);
      for (let i = 0; i < bufferSize; i++) {
        data[i] = Math.random() * 2 - 1;
      }
      return buffer;
    }

        }

        playNote(ctx, 110, now, 1.6, 'sawtooth', 0.28, 1760);

    function toggleLoop(loopName) {
      const ctx = getAudioContext();
      if (activeLoops[loopName]) {
        stopSingleLoop(loopName);
        return;
      }

      const master = getMasterGain();
      const buffer = createNoiseBuffer(ctx, 4);
      const src = ctx.createBufferSource();
      src.buffer = buffer;
      src.loop = true;

      const filter = ctx.createBiquadFilter();
      const gain = ctx.createGain();
      let extraOsc = null;
      let waveInterval = null;

      if (loopName === 'rain') {
        filter.type = 'lowpass';
        filter.frequency.value = 850;
        gain.gain.value = 0.22 * master;
      } else if (loopName === 'wind') {
        filter.type = 'bandpass';
        filter.frequency.value = 340;
        filter.Q.value = 4.5;
        gain.gain.value = 0.3 * master;
      } else if (loopName === 'forest') {
        filter.type = 'bandpass';
        filter.frequency.value = 1400;
        filter.Q.value = 1.5;
        gain.gain.value = 0.12 * master;
      } else if (loopName === 'classroom') {
        filter.type = 'bandpass';
        filter.frequency.value = 600;
        filter.Q.value = 2;
        gain.gain.value = 0.2 * master;
      } else if (loopName === 'ocean') {
        filter.type = 'lowpass';
        filter.frequency.value = 420;
        gain.gain.value = 0.25 * master;
        waveInterval = setInterval(() => {
          if (!activeLoops.ocean) return;
          const t = ctx.currentTime;
          gain.gain.setValueAtTime(0.12 * getMasterGain(), t);
          gain.gain.linearRampToValueAtTime(0.32 * getMasterGain(), t + 1.8);
          gain.gain.linearRampToValueAtTime(0.12 * getMasterGain(), t + 3.6);
        }, 3800);
      } else if (loopName === 'space_ship') {
        filter.type = 'lowpass';
        filter.frequency.value = 180;
        gain.gain.value = 0.28 * master;
        extraOsc = ctx.createOscillator();
        extraOsc.type = 'sine';
        extraOsc.frequency.value = 65;
        const oscGain = ctx.createGain();
        oscGain.gain.value = 0.18 * master;
        extraOsc.connect(oscGain);
        oscGain.connect(ctx.destination);
        extraOsc.start();
      }

      src.connect(filter);
      filter.connect(gain);
      gain.connect(ctx.destination);
      src.start();

      let birdInterval = null;
      if (loopName === 'forest') {
        birdInterval = setInterval(() => {
          if (!activeLoops.forest) return;
          const t = ctx.currentTime;
          playNote(ctx, 2200 + Math.random() * 800, t, 0.08, 'sine', 0.14);
          playNote(ctx, 2600 + Math.random() * 600, t + 0.11, 0.09, 'sine', 0.14);
        }, 2300);
      }

      activeLoops[loopName] = { src, extraOsc, birdInterval, waveInterval };
      setMixerVisualStatus(`Ambiente: ${loopName}`, 0);
      filterCabinaSfx();
    }

    function stopSingleLoop(loopName) {
      const item = activeLoops[loopName];
      if (!item) return;
      try { item.src.stop(); } catch (e) {}
      if (item.extraOsc) { try { item.extraOsc.stop(); } catch (e) {} }
      if (item.birdInterval) clearInterval(item.birdInterval);
      if (item.waveInterval) clearInterval(item.waveInterval);
      delete activeLoops[loopName];
      if (Object.keys(activeLoops).length === 0) setMixerVisualStatus(null);
      filterCabinaSfx();
    }

        stopSlotAudio(i);

      setMixerVisualStatus(null);
    }

    function loadSlotLocalFile(slotId, event) {
      const file = event.target.files[0];
      if (!file) return;
      const url = URL.createObjectURL(file);
      setupSlotAudio(slotId, url, `Archivo: ${file.name}`, file.name.replace(/\.[^/.]+$/, ''));
      event.target.value = '';
    }

    function applySlotUrl(slotId) {
      const input = document.getElementById(`slot-url-input-${slotId}`);
      const url = input ? input.value.trim() : '';
      if (!url) {
        showToast('Introduce un enlace URL válido');
        return;
      }
      audioSlots[slotId].urlInputOpen = false;
      const shortName = url.split('/').pop().split('?')[0] || `Enlace Cajón ${slotId}`;
      setupSlotAudio(slotId, url, `URL: ${url}`, shortName);
    }

    function setupSlotAudio(slotId, url, sourceInfo, suggestedTitle) {
      const slot = audioSlots[slotId];
      if (slot.audio) {
        slot.audio.pause();
      }
      const audio = new Audio(url);
      audio.volume = slot.volume;

      audio.onplay = () => {
        setMixerVisualStatus(slot.customName, 0);
        renderSixSlots();
      };
      audio.onpause = () => {
        if (Object.keys(activeLoops).length === 0) setMixerVisualStatus(null);
        renderSixSlots();
      };
      audio.onended = () => {
        if (Object.keys(activeLoops).length === 0) setMixerVisualStatus(null);
        renderSixSlots();
      };
      audio.onerror = () => {
        showToast(`No se pudo cargar el audio en Cajón ${slotId}. Si es un enlace externo, descárgalo a tu PC.`);
      };

      slot.audio = audio;
      slot.sourceInfo = sourceInfo;
      if (slot.customName === DEFAULT_SLOT_LABELS[slotId - 1] && suggestedTitle) {
        slot.customName = `Cajón ${slotId} · ${suggestedTitle}`;
      }

      renderSixSlots();
      showToast(`Audio cargado en Cajón ${slotId}`);
    }

    function toggleSlotPlay(slotId) {
      const slot = audioSlots[slotId];
      if (!slot || !slot.audio) return;
      if (slot.audio.paused) {
        slot.audio.play().catch(() => {
          showToast(`Bloqueo CORS en el enlace del Cajón ${slotId}. Descarga el archivo y cárgalo localmente.`);
        });
      } else {
        slot.audio.pause();
      }
      renderSixSlots();
    }

    function stopSlotAudio(slotId) {
      const slot = audioSlots[slotId];
      if (!slot || !slot.audio) return;
      slot.audio.pause();
      slot.audio.currentTime = 0;
      renderSixSlots();
    }

    function setSlotVolume(slotId, val) {
      const slot = audioSlots[slotId];
      if (!slot) return;
      const clamped = Math.min(1, Math.max(0, parseFloat(val)));
      slot.volume = clamped;
      if (slot.audio) slot.audio.volume = clamped;
      const lbl = document.getElementById(`slot-vol-lbl-${slotId}`);
      const slider = document.getElementById(`slot-vol-slider-${slotId}`);
      if (lbl) lbl.textContent = `${Math.round(clamped * 100)}%`;
      if (slider) slider.value = clamped;
    }

    function fadeSlotVolume(slotId, targetVol) {
      const slot = audioSlots[slotId];
      if (!slot || !slot.audio) return;
      if (slot.fadeInterval) clearInterval(slot.fadeInterval);

      slot.fadeInterval = setInterval(() => {
        const diff = targetVol - slot.volume;
        if (Math.abs(diff) < 0.03) {
          setSlotVolume(slotId, targetVol);
          clearInterval(slot.fadeInterval);
        } else {
          setSlotVolume(slotId, +(slot.volume + diff * 0.25).toFixed(2));
        }
      }, 45);
    }

    function updateScriptStats() {
      const text = document.getElementById('script-editor').value.trim();
      const spoken = text.replace(/\[.*?\]/g, ' ').trim();
      const words = spoken ? spoken.split(/\s+/).length : 0;

    function renderTeleprompter() {
      const raw = document.getElementById('script-editor').value;
      const box = document.getElementById('teleprompter-box');
      const lines = raw.split('\n');

      const speakerPalette = [
        'text-sky-400',
        'text-emerald-400',
        'text-purple-400',
        'text-pink-400',
        'text-amber-300'
      ];
      const speakerColors = {};
      let colorIdx = 0;

      const contentHtml = lines.map(line => {
        const trimmed = line.trim();
        if (!trimmed) return '<div class="h-4"></div>';

        if (trimmed.startsWith('[') && trimmed.endsWith(']')) {
          return `<div class="px-4 py-2 rounded-xl theme-accent-bg font-extrabold text-center shadow text-base sm:text-lg tracking-wide">${escapeHtml(trimmed)}</div>`;
        }

        const colonIdx = trimmed.indexOf(':');
        if (colonIdx > 0 && colonIdx < 28) {
          const speaker = trimmed.slice(0, colonIdx + 1);
          const speech = trimmed.slice(colonIdx + 1);
          if (!speakerColors[speaker]) {
            speakerColors[speaker] = speakerPalette[colorIdx % speakerPalette.length];
            colorIdx++;
          }
          return `<p class="leading-relaxed"><span class="font-extrabold ${speakerColors[speaker]} underline decoration-2 mr-2">${escapeHtml(speaker)}</span>${escapeHtml(speech)}</p>`;
        }

        return `<p class="leading-relaxed">${escapeHtml(trimmed)}</p>`;
      }).join('');

      box.innerHTML = contentHtml + '<div class="h-72"></div>';
    }

    function prompterStep(timestamp) {
      if (!isPrompterScrolling) return;
      const box = document.getElementById('teleprompter-box');

      if (lastFrameTime !== null) {
        const dt = (timestamp - lastFrameTime) / 1000;
        const speed = parseInt(document.getElementById('prompter-speed').value, 10);
        const pxPerSec = 10 + speed * 15;
        exactScrollPos += pxPerSec * dt;
        box.scrollTop = exactScrollPos;

        if (box.scrollTop + box.clientHeight >= box.scrollHeight - 2) {
          stopPrompterScroll();
          showToast('Fin del guion alcanzado');
          return;
        }
      }

      lastFrameTime = timestamp;
      prompterRafId = requestAnimationFrame(prompterStep);
    }

    function togglePrompterScroll() {
      if (isPrompterScrolling) {
        stopPrompterScroll();
      } else {
        const box = document.getElementById('teleprompter-box');
        if (box.scrollTop + box.clientHeight >= box.scrollHeight - 10) {
          box.scrollTop = 0;
        }
        exactScrollPos = box.scrollTop;
        lastFrameTime = null;
        isPrompterScrolling = true;

        const btn = document.getElementById('prompter-play-btn');
        btn.innerHTML = '<span>⏸️</span> <span>Pausar</span>';
        btn.classList.replace('bg-emerald-600', 'bg-amber-600');
        prompterRafId = requestAnimationFrame(prompterStep);
      }
    }

    function stopPrompterScroll() {
      isPrompterScrolling = false;
      if (prompterRafId) cancelAnimationFrame(prompterRafId);
      prompterRafId = null;
      lastFrameTime = null;
      const btn = document.getElementById('prompter-play-btn');
      btn.innerHTML = '<span>▶️</span> <span>Iniciar Desplazamiento</span>';
      btn.classList.replace('bg-amber-600', 'bg-emerald-600');
    }

    function resetPrompterScroll() {
      stopPrompterScroll();
      const box = document.getElementById('teleprompter-box');
      box.scrollTop = 0;
      exactScrollPos = 0;
    }

    function speakScript() {
      if (!('speechSynthesis' in window)) {
        showToast('Tu navegador no soporta lectura en voz alta.');
        return;
      }
      if (window.speechSynthesis.speaking) {
        window.speechSynthesis.cancel();
        showToast('Lectura en voz alta detenida');
        return;
      }
      const raw = document.getElementById('script-editor').value;
      const cleanText = raw.replace(/\[.*?\]/g, '');
      const utterance = new SpeechSynthesisUtterance(cleanText);
      utterance.lang = 'es-ES';
      utterance.rate = 0.92;
      window.speechSynthesis.speak(utterance);
      showToast('Reproduciendo lectura en voz alta (Apoyo DUA)...');
    }

    function exportScriptPDF() {
      const title = document.getElementById('script-title').value || 'Guion de Radio';
      const stats = document.getElementById('word-stats').textContent;
      const raw = document.getElementById('script-editor').value;
      const today = new Date().toLocaleDateString('es-ES');

      const formattedBody = raw.split('\n').map(line => {
        const t = line.trim();
        if (!t) return '<br/>';
        if (t.startsWith('[') && t.endsWith(']')) {
          return `<div style="background:#fef3c7; border:1px solid #d97706; padding:6px 12px; margin:10px 0; font-weight:bold; border-radius:6px;">${escapeHtml(t)}</div>`;
        }
        return `<p style="margin:8px 0; font-size:13pt; line-height:1.6;">${escapeHtml(t)}</p>`;
      }).join('');

      const printArea = document.getElementById('print-area');
      printArea.innerHTML = `
        <div style="border-bottom: 3px solid #d97706; padding-bottom: 12px; margin-bottom: 20px;">
          <div style="font-size: 10pt; font-weight: bold; color: #b45309; text-transform: uppercase;">
            Radio Ad-Gemellas · CEIP San José (Palenciana)
          </div>
          <h1 style="font-size: 18pt; font-weight: 800; margin: 4px 0;">${escapeHtml(title)}</h1>
          <div style="font-size: 10pt; color: #475569;">Fecha: ${today} · Duración estimada: ${stats}</div>
        </div>
        <div>${formattedBody}</div>
      `;

      printArea.classList.remove('hidden');
      window.print();
      printArea.classList.add('hidden');
    }

    let isOnAir = false;
    let timerSeconds = 0;
    let timerInterval = null;
    let isTimerRunning = false;

    function toggleTimer() {
      const btn = document.getElementById('timer-play-btn');
      if (isTimerRunning) {
        clearInterval(timerInterval);
        isTimerRunning = false;
        btn.textContent = '▶️';
      } else {
        isTimerRunning = true;
        btn.textContent = '⏸️';
        timerInterval = setInterval(() => {
          timerSeconds++;
          const m = String(Math.floor(timerSeconds / 60)).padStart(2, '0');
          const s = String(timerSeconds % 60).padStart(2, '0');
          document.getElementById('timer-display').textContent = `${m}:${s}`;
        }, 1000);
      }
    }

    function resetTimer() {
      clearInterval(timerInterval);
      isTimerRunning = false;
      timerSeconds = 0;
      document.getElementById('timer-display').textContent = '00:00';
      document.getElementById('timer-play-btn').textContent = '▶️';
    }

    function toggleOnAir() {
      isOnAir = !isOnAir;
      const btn = document.getElementById('on-air-btn');
      const dot = document.getElementById('on-air-dot');
      const text = document.getElementById('on-air-text');

      if (isOnAir) {
        btn.className = 'px-3.5 py-1.5 rounded-xl font-extrabold text-xs tracking-wider uppercase border-2 border-red-400 bg-red-600 text-white shadow-lg on-air-active flex items-center gap-2 cursor-pointer';
        dot.className = 'w-2.5 h-2.5 rounded-full bg-white';
        text.textContent = 'EN EL AIRE';
        if (!isTimerRunning) toggleTimer();
        showToast('Piloto EN EL AIRE activado');
      } else {
        btn.className = 'px-3.5 py-1.5 rounded-xl font-extrabold text-xs tracking-wider uppercase border-2 border-slate-500 theme-card transition-all flex items-center gap-2 shadow cursor-pointer';
        dot.className = 'w-2.5 h-2.5 rounded-full bg-slate-500';
        text.textContent = 'FUERA DEL AIRE';
        if (isTimerRunning) toggleTimer();
      }
    }

    function copyFilename() {

      const temp = document.createElement('input');

    function escapeHtml(str) {
      return String(str)
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;');
    }

    function showToast(message) {
      const container = document.getElementById('toast-container');
      const toast = document.createElement('div');
      toast.className = 'bg-slate-900/95 border border-amber-500/60 text-white text-xs font-bold px-4 py-2.5 rounded-xl shadow-xl transition-all duration-300';
      toast.textContent = message;
      container.appendChild(toast);
      setTimeout(() => {
        toast.style.opacity = '0';
        setTimeout(() => toast.remove(), 300);
      }, 2800);
    }
